cmake_minimum_required(VERSION 3.24)
project(LocalForwarder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(yyjson STATIC ../third_party/yyjson/yyjson.c)
target_include_directories(yyjson PUBLIC ../third_party/yyjson)
add_library(lf_core STATIC src/core.cpp)
target_include_directories(lf_core PUBLIC src)
target_link_libraries(lf_core PUBLIC yyjson)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  target_compile_options(lf_core PRIVATE /utf-8 /W4 /permissive-)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(lf_core_tests tests/core_tests.cpp)
  target_link_libraries(lf_core_tests PRIVATE lf_core)
  add_test(NAME core COMMAND lf_core_tests)
endif()
if(WIN32)
  add_library(lf_platform STATIC src/platform.cpp src/auth.cpp)
  target_compile_definitions(lf_platform PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(lf_platform PUBLIC lf_core shell32 ole32 advapi32 uuid)
  add_executable(LocalForwarder WIN32 src/main.cpp app.rc)
  target_link_libraries(LocalForwarder PRIVATE lf_platform comctl32 user32 gdi32)
  target_include_directories(LocalForwarder PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  if(MSVC)
    target_link_options(LocalForwarder PRIVATE /MANIFEST:NO)
  endif()
  if(BUILD_TESTING)
    add_executable(lf_platform_tests tests/platform_tests.cpp)
    target_link_libraries(lf_platform_tests PRIVATE lf_platform)
    add_test(NAME platform COMMAND lf_platform_tests)
  endif()
endif()
